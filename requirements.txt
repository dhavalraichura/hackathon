# ─────────────────────────────────────────────
# Enterprise Knowledge Assistant — Requirements
# ─────────────────────────────────────────────

# Core LLM & Embeddings
sentence-transformers       # Open-source embeddings (HuggingFace)
transformers                # HuggingFace transformer models
torch                       # PyTorch backend

# Vector Database
faiss-cpu                   # Facebook FAISS - local vector search

# Document Processing
pypdf                       # PDF text extraction
pdfplumber                  # PDF table + layout extraction
pdf2image                   # Convert PDF pages to images for OCR
beautifulsoup4              # HTML/wiki parsing
requests                    # HTTP requests for fetching docs
pandas                      # CSV ticket processing
python-docx                 # Word document (.docx) support
openpyxl                    # Excel (.xlsx) support
imageio-ffmpeg              # Bundled ffmpeg (Windows-friendly, also works on Linux/Mac)

# OCR
pytesseract                 # OCR engine wrapper (needs tesseract-ocr system package)
Pillow                      # Image processing for OCR pipeline

# Video Transcription
openai-whisper              # On-premise audio/video transcription (Whisper)
moviepy                     # Video → audio extraction before Whisper

# LLM (local open-source via Ollama OR API)
ollama                      # Run Gemma4/LLaMA locally (recommended)
openai                      # Optional: OpenAI API fallback
anthropic                   # Optional: Claude API fallback

# Web UI
streamlit                   # Chat interface

# API
fastapi                     # REST API
uvicorn                     # ASGI server

# Evaluation
scikit-learn                # Precision, Recall, F1
numpy                       # Numerical ops
rouge-score                 # Text similarity scoring

# Utilities
tqdm                        # Progress bars
python-dotenv               # Environment variables
colorama                    # Colored terminal output
tabulate                    # Pretty table printing
